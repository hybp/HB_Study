### !!! 비동기 처리 확인하는법

테스트 하고 로그를 보면 된다
동일한 Thread로 계속 처리하고 있다면 Sync
계속 새로운 Thread로 실행된다면 Async


### 테스트 만들기

예시)
```java
public class BalanceControllerTest {  
    private BalanceController balanceController;  
    private AccountService accountService;  
    private Database db;  
    private final Long accountId = 1L;  
  
    @BeforeEach  
    void setUp() {  
        db = new Database();  
        accountService = new AccountService(db);  
        balanceController = new BalanceController(db, accountService);  
        db.balance(accountId, 100L);  
    }  
  
    @Test  
    void testConcurrentDeposits() throws InterruptedException {  
        // 테스트 파라미터 설정  
        int numThreads = 5;  
        long depositAmount = 10L;  
  
        // 스레드 풀과 CountDownLatch 생성  
        ExecutorService executorService = Executors.newFixedThreadPool(numThreads);  
        CountDownLatch startLatch = new CountDownLatch(1);  
        CountDownLatch finishLatch = new CountDownLatch(numThreads);  
        AtomicInteger successCount = new AtomicInteger(0);  
  
        // 동시 입금 시도  
        for (int i = 0; i < numThreads; i++) {  
            executorService.submit(() -> {  
                try {  
                    // 스레드 시작 대기  
                    startLatch.await();  
  
                    // 입금 작업  
                    balanceController.deposit(accountId, depositAmount);  
                    successCount.incrementAndGet();  
                } catch (Exception e) {  
                    // 입금 실패 시 메시지 출력  
                    System.out.println("Deposit failed: " + e.getMessage());  
                } finally {  
                    finishLatch.countDown();  
                }  
            });  
        }  
  
        // 모든 스레드 시작  
        startLatch.countDown();  
        // 모든 스레드 종료 대기  
        finishLatch.await();  
        executorService.shutdown();  
  
  
        // 결과 출력  
        System.out.println("입금 성공 횟수: " + successCount.get());  
        System.out.println("최종 잔고: " + balanceController.balance(accountId).getBalance());  
        System.out.println("최종 잔고 정답: ");  
    }  
}

```
